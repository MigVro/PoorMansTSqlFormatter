WITH XMLNAMESPACES (
	'http://a.com/x' AS ns1
	,DEFAULT 'http://b.com/y'
	)
SELECT a.Id AS [ns1:Id]
	,a.Name
FROM dbo.A a
FOR XML PATH('Row')
	,ROOT('Rows');

	,cte

	)

FROM cte;

	,b
