/* implicit aliases get an explicit AS; anything unclear is left alone */
SELECT a.x foo, a.y AS bar, 'lit' lbl, COUNT(*) cnt, (SELECT 1) sub,
    CASE WHEN a.z = 1 THEN 2 END c2, a.x + 1 [sum], @v = 3, alias2 = a.col,
    SUM(q) OVER (PARTITION BY p) w, a.b, 5 five
FROM dbo.T a
    INNER JOIN (SELECT 1 q) d ON a.id = d.q
    LEFT JOIN dbo.U u2 WITH (NOLOCK) ON u2.id = a.id
    CROSS APPLY dbo.F(a.id) f1,
    dbo.V v1;
